{{- /* Start/end times and the big-date label. Start and end both at 00:00 = date-only (allDay). */ -}} {{- $start := .Date -}} {{- $end := $start -}} {{- with .Params.end -}}{{- $end = time.AsTime . -}}{{- end -}} {{- $allDay := and (eq ($start.Format "15:04:05") "00:00:00") (eq ($end.Format "15:04:05") "00:00:00") -}} {{- $until := $end -}} {{- if $allDay }}{{ $until = $end.AddDate 0 0 1 }}{{ end -}} {{- $sameDay := eq ($start.Format "2006-01-02") ($end.Format "2006-01-02") -}} {{- $sameMonth := eq ($start.Format "2006-01") ($end.Format "2006-01") -}} {{- $sameYear := eq $start.Year $end.Year -}} {{- $label := $start | time.Format (T "date_full") -}} {{- if not $sameDay -}} {{- $to := $end | time.Format (T "date_full") -}} {{- if $sameMonth -}} {{- /* word joiner: no line break after the dash */ -}} {{- $label = printf "%s–\u2060%s" ($start | time.Format (T "date_day")) $to -}} {{- else if $sameYear -}} {{- $label = printf "%s\u00a0– %s" ($start | time.Format (T "date_day_month")) $to -}} {{- else -}} {{- $label = printf "%s\u00a0– %s" $label $to -}} {{- end -}} {{- end -}} {{- return dict "start" $start "end" $end "until" $until "allDay" $allDay "sameDay" $sameDay "label" $label "iso" ($until.Format "2006-01-02T15:04:05-07:00") "isPast" ($until.Before now) -}}
